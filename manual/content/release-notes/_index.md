<script>
(function(){
  document.querySelectorAll('.rn-tabs').forEach(function(wrap){
    var tabs = wrap.querySelectorAll('[data-rn-tab]');
    tabs.forEach(function(tab){
      tab.addEventListener('click', function(){
        var target = tab.getAttribute('data-rn-tab');
        tabs.forEach(function(t){
          var active = t === tab;
          t.classList.toggle('on', active);
          t.setAttribute('aria-selected', active ? 'true' : 'false');
        });
        wrap.querySelectorAll('[data-rn-panel]').forEach(function(panel){
          panel.classList.toggle('on', panel.getAttribute('data-rn-panel') === target);
        });
      });
    });
  });
})();
</script>
